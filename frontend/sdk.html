# Node (Express) example
import { AICreditClient } from "billbridge-ai-credit-sdk";
const billbridge = new AICreditClient({
  apiKey: process.env.AI_CREDIT_API_KEY,
  baseUrl: "https://api.billbridge.in/v1",
});

app.post("/run-model", async (req, res) => {
  const balance = await billbridge.credits.balance();
  if (balance.balance < 10) return res.status(402).json({ error: "low credits" });

  // run your LLM call here...
  const result = { ok: true };

  await billbridge.usage.record({
    event_id: `evt_${Date.now()}`,
    model: "gpt-4.1",
    input_token: 1200,
    output_token: 400,
  });
  res.json(result);
});

          

          

            
# Python (FastAPI) example
from fastapi import FastAPI, HTTPException
from ai_credit_sdk.client import AICreditClient
import time

app = FastAPI()
billbridge = AICreditClient(
    api_key="YOUR_API_KEY",
    base_url="https://api.billbridge.in/v1",
)

@app.post("/run-model")
def run_model():
    bal = billbridge.credits.balance()
    if bal["balance"] < 10:
        raise HTTPException(status_code=402, detail="low credits")

    # run your LLM call here...
    billbridge.usage.record({
        "event_id": f"evt_{int(time.time())}",
        "model": "gpt-4.1",
        "input_token": 1000,
        "output_token": 500,
    })
    return {"ok": True}
